Java Code – CC102 Final Performance Task

Console-Based Number Guessing Game

This is my final Performance Task for CC102 – Object-Oriented Programming. The game generates a random number between 1 and 100 and challenges the player to guess it within a limited number of attempts. It demonstrates the use of classes, methods, loops, conditionals, and user input handling in Java. 


import java.util.Scanner;
import java.util.Random;

public class SimpleGame {

    public static void main(String[] args) {

        Scanner scanner = new Scanner(System.in);
        Random random = new Random();

        int secretNumber = random.nextInt(10) + 1;
        int guess = 0;
        int attempts = 0;

        System.out.println("=== Guess the Number (1-10) ===");

        while (guess != secretNumber) {
            System.out.print("Enter your guess: ");
            guess = scanner.nextInt();
            attempts++;

            if (guess < secretNumber) {
                System.out.println("Too low! Try again.");
            } else if (guess > secretNumber) {
                System.out.println("Too high! Try again.");
            } else {
                System.out.println("Correct! You got it in " + attempts + " tries!");
            }
        }

        scanner.close();
    }
}
